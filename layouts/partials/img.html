{{- /* Responsive image. Context: dict src (required) path as stored in content, e.g. "/img/foo.png" alt (required) alt text sizes (optional) the sizes attribute, defaults to a sensible layout default class (optional) class attribute eager (optional) true to load eagerly (LCP images: hero, first gallery image) width (optional) intrinsic width hint for CSS Variants are generated by tools/optimize_images.py into static/img/_gen/-.webp. Falls back to the original file if no variants exist. */ -}} {{- $src := .src -}} {{- $alt := .alt | default "" -}} {{- $class := .class | default "" -}} {{- $sizes := .sizes | default "(max-width: 600px) 100vw, (max-width: 1000px) 50vw, 33vw" -}} {{- $eager := .eager | default false -}} {{- $clean := strings.TrimPrefix "/" $src -}} {{- $dir := path.Dir $clean -}} {{- $base := path.Base $clean -}} {{- $name := strings.TrimSuffix (path.Ext $base) $base -}} {{- $genDir := path.Join $dir "_gen" -}} {{- /* Candidates: the standard widths plus the source's own width, which the optimizer also emits (never upscales past the source). */ -}} {{- /* Real widths come from the manifest the optimizer writes; fall back to the standard set if it is missing (e.g. before the first build). */ -}} {{- $widths := slice "400" "800" "1200" "1600" "2000" -}} {{- $mPath := path.Join "static" $genDir "manifest.json" -}} {{- if fileExists $mPath -}} {{- $manifest := (os.ReadFile $mPath | transform.Unmarshal) -}} {{- with index $manifest $name -}} {{- $widths = slice -}} {{- range .widths }}{{ $widths = $widths | append (printf "%v" .) }}{{ end -}} {{- end -}} {{- end -}} {{- $webp := slice -}} {{- $webpSrcset := slice -}} {{- $jpgSrcset := slice -}} {{- $hasJpg := false -}} {{- range $w := $widths -}} {{- $wp := path.Join $genDir (printf "%s-%s.webp" $name $w) -}} {{- if fileExists (path.Join "static" $wp) -}} {{- $url := $wp | relURL -}} {{- $webpSrcset = $webpSrcset | append (printf "%s %sw" $url $w) -}} {{- $jp := path.Join $genDir (printf "%s-%s.jpg" $name $w) -}} {{- if fileExists (path.Join "static" $jp) -}} {{- $hasJpg = true -}} {{- $jpgSrcset = $jpgSrcset | append (printf "%s %sw" ($jp | relURL) $w) -}} {{- end -}} {{- end -}} {{- end -}} {{- $origURL := $src | relURL -}} {{- $w := .width -}} {{- $loading := cond $eager "eager" "lazy" -}} {{- $fetchprio := cond $eager "high" "auto" -}} {{- if gt (len $webpSrcset) 0 -}} {{- if $hasJpg -}} {{- else -}} {{- end -}} {{- else -}} {{- end -}}
